# -------------------- FILE DISCOVERY --------------------

def locate_file(filename, preferred_folder=None):
    candidates = []

    if preferred_folder:
        candidates.append(BASE_DIR / preferred_folder / filename)

    candidates.extend([
        BASE_DIR / filename,
        BASE_DIR / "data" / "raw" / filename,
        BASE_DIR / "models" / filename,
        BASE_DIR.parent / filename,
        Path.home() / "Downloads" / filename,
        Path.home() / "Desktop" / filename,
        Path.home() / "Documents" / filename,
    ])

    seen = set()

    for path in candidates:
        try:
            resolved = path.resolve()

            if resolved in seen:
                continue

            seen.add(resolved)

            if resolved.is_file():
                return resolved

        except (OSError, RuntimeError):
            pass

    # Search within this project only.
    try:
        for path in BASE_DIR.rglob(filename):
            if path.is_file():
                return path.resolve()
    except OSError:
        pass

    return None

# -------------------- LOAD DATA --------------------

@st.cache_data
def load_data(path, modified):
    data = pd.read_csv(path)

    required = {
        "Date", "Order_ID", "Product", "Category",
        "Quantity", "Total_Sales", "Order_Status"
    }

    missing = required - set(data.columns)

    if missing:
        raise ValueError(
            f"Missing required CSV columns: {sorted(missing)}"
        )

    data["Date"] = pd.to_datetime(
        data["Date"], errors="coerce"
    )

    for col in ["Quantity", "Total_Sales"]:
        data[col] = pd.to_numeric(
            data[col], errors="coerce"
        )

    data = data.dropna(
        subset=["Date", "Product", "Quantity", "Total_Sales"]
    ).copy()

    for col in ["Category", "Product", "Order_Status"]:
        data[col] = data[col].astype(str).str.strip()

    return data


@st.cache_resource
def load_model(path, modified):
    return joblib.load(path)
